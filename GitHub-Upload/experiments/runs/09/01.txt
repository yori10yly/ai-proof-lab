constructor
· calc
    w * a = k * a + (w - k) * a := by rw [← Nat.add_mul, h]
    _ ≤ _ := Nat.add_le_add_left (Nat.mul_le_mul_left (w - k) hab) _

    k * a + (w - k) * b ≤ k * b + (w - k) * b :=
      Nat.add_le_add_right (Nat.mul_le_mul_left k hab) _
    _ = w * b := by rw [← Nat.add_mul, h]
